cmake_minimum_required(VERSION 3.22.1)
project(hoenn_android C)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(LIBMGBA_ONLY ON CACHE BOOL "" FORCE)
set(M_CORE_GB OFF CACHE BOOL "" FORCE)
set(M_CORE_GBA ON CACHE BOOL "" FORCE)
set(USE_DEBUGGERS OFF CACHE BOOL "" FORCE)
set(USE_GDB_STUB OFF CACHE BOOL "" FORCE)
set(ENABLE_SCRIPTING OFF CACHE BOOL "" FORCE)
if(DEFINED ENV{HOENN_MGBA_SOURCE} AND NOT "$ENV{HOENN_MGBA_SOURCE}" STREQUAL "")
    set(MGBA_SOURCE "$ENV{HOENN_MGBA_SOURCE}")
else()
    set(MGBA_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../../../../../.local/mgba")
endif()
add_subdirectory("${MGBA_SOURCE}" "${CMAKE_BINARY_DIR}/mgba" EXCLUDE_FROM_ALL)
add_library(hoenn SHARED emulator.c)
target_include_directories(hoenn PRIVATE "${MGBA_SOURCE}/include" "${CMAKE_BINARY_DIR}/mgba/include")
target_link_libraries(hoenn mgba android log jnigraphics)
